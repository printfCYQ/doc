/**
 * @param {number[]} nums
 * @return {number}
 */
var minOperations = function (nums) {
    let res = 0;
    for (let i = 1; i < nums.length; i++) {
        // 如果当前数小于等于上一个数
        if (nums[i] <= nums[i - 1]) {
            res += nums[i - 1] + 1 - nums[i]; // 计划两个数的差再加1（要比上个数大）
            nums[i] = nums[i - 1] + 1; // 更新当前数的值
        }
    }
    return res;
};
