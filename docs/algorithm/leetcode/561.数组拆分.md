### sort

-   从小到大排序后，只取奇数位置的数（满足题意）

```javascript
/**
 * @param {number[]} nums
 * @return {number}
 */
var arrayPairSum = function (nums) {
    nums.sort((a, b) => a - b);
    let res = 0;
    for (let i = 0, len = nums.length; i < len; i = i + 2) {
        res += nums[i];
    }
    return res;
};
```
