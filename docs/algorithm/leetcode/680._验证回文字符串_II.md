/**
 * @param {string} s
 * @return {boolean}
 */
var validPalindrome = function (s) {
    let low = 0;
    let high = s.length - 1;
    if (checkPalindrome(s, low, high)) return true; // 是回文串直接返回true
    while (low < high) {
        if (s[low] !== s[high]) { // 当不相等时： 跳过这个字符；（两种跳过办法） 。判断剩余的字符串（s[low+1, high] || s[low, high-1]）是否是回文串。其中一个可以形成回文串就返回true
            return checkPalindrome(s, low, high - 1) || checkPalindrome(s, low + 1, high);
        }
        low++;
        high--;
    }
};
// 判断是否为回文串
var checkPalindrome = function (s, low, high) {
    while (low < high) {
        if (s[low] !== s[high]) {
            return false;
        }
        low++;
        high--;
    }
    return true;
};
