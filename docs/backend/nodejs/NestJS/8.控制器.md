## GET 请求

### @Request

```typescript
import { Controller, Get, Request } from '@nestjs/common';
import { BookService } from './book.service';

@Controller('book')
export class BookController {
  constructor(private readonly bookService: BookService) {}

  @Get()
  findAll(@Request() req) {
    console.log(req);
    return {
      code: 200,
    };
  }
}

```




> 直接读取指定参数

```typescript

    };
  }

```

## 动态路由

### @Request

```typescript
import { Controller, Get, Request } from '@nestjs/common';
import { BookService } from './book.service';

@Controller('book')
export class BookController {
  constructor(private readonly bookService: BookService) {}

  @Get(':id')
  findAll(@Request() req) {
    console.log(req);
    return {
      code: 200,
    };
  }
}
```



### @Param

## @HttpCode

```typescript
import { Controller, Get, HttpCode } from '@nestjs/common';
import { BookService } from './book.service';

@Controller('book')
export class BookController {
  constructor(private readonly bookService: BookService) {}

  @Get()
  @HttpCode(404)
  findAll() {
    return {
      code: 200,
    };
  }
}
```
