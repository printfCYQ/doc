# 本地读取视频任意时段画面截图

```html
<!DOCTYPE html>
<html lang="en">

  <head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>本地读取视频任意时段画面截图</title>
    <style>
      img {
        width: 100px;
        height: 66px;
      }
    </style>
  </head>

  <body>
    <input type="file">
    <script>
      const inp = document.querySelector('input[type="file"]');

      // 当文件选择框的值改变时触发事件
      inp.onchange = async (e) => {
        const file = e.target.files[0];

        for (let i = 0; i < 20; i++) {
          const frame = await captureFrame(file, i);
          createPreview(frame);
        }
      }

      // 截取指定时间的视频帧
      async function captureFrame(videoFile, time = 0) {
        const vdo = document.createElement('video');
        vdo.currentTime = time;
        vdo.muted = true;
        vdo.autoplay = true;

        // 创建一个 Promise，用于等待视频能够播放
        const canPlayPromise = new Promise((resolve) => {
          vdo.oncanplay = resolve;
        });

        vdo.src = URL.createObjectURL(videoFile);
        await canPlayPromise;

        const frame = await drawVideo(vdo);
        return frame;
      }

      // 绘制视频帧
      function drawVideo(vdo) {
        return new Promise((resolve) => {
          const cvs = document.createElement('canvas');
          const ctx = cvs.getContext('2d');
          cvs.width = vdo.videoWidth;
          cvs.height = vdo.videoHeight;
          ctx.drawImage(vdo, 0, 0, cvs.width, cvs.height);
          cvs.toBlob((blob) => {
            resolve({
              blob,
              url: URL.createObjectURL(blob)
            });
          });
        });
      }

      // 创建预览并显示截取的视频帧
      function createPreview(frame) {
        const img = document.createElement('img');
        img.src = frame.url;
        document.body.appendChild(img);
      }
    </script>
  </body>

</html>

```
