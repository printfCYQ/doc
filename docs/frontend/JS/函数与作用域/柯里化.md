# 柯里化

```javascript
function currying(fn, ...bindArgs) {
  return (...args) => {
    const allArgs = [...bindArgs, ...args];
    return allArgs.length < fn.length
      ? currying(fn, ...allArgs)
      : fn(...allArgs);
  };
}

const sum = (a, b, c, d) => a + b + c + d;

console.log(currying(sum)(1)(2)(3)(4));
console.log(currying(sum, 1)(2)(3)(4));
console.log(currying(sum, 1, 2)(3)(4));
console.log(currying(sum, 1)(2, 3)(4));
console.log(currying(sum, 1, 2)(3, 4));
```
