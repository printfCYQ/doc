# 大驼峰-小驼峰

## 大驼峰->小驼峰

```javascript
function pascalToCamel(pascalCase) {
    return pascalCase.charAt(0).toLowerCase() + pascalCase.slice(1);;
}
console.log(pascalToCamel("helloworld")); // helloworld
console.log(pascalToCamel("HelloWorld")); // helloWorld
console.log(pascalToCamel("HelloWorld123")); // helloWorld123
```

## 大驼峰->连字符

```javascript
const camelToKebab = (str, separator) => {
  return str.replace(/([a-z])([A-Z])/g, `$1${separator}$2`).toLowerCase();
};

const camelCase = 'myVariableName';
const kebabCase = camelToKebab(camelCase);

console.log(kebabCase); // 输出: my-variable-name
```

## 小驼峰->大驼峰

```javascript
function camelToPascal(camelCase) {
    return camelCase.charAt(0).toUpperCase() + camelCase.slice(1);;
}
console.log(camelToPascal("helloworld")); // Helloworld
console.log(camelToPascal("helloWorld")); // HelloWorld
console.log(camelToPascal("helloWorld123")); // HelloWorld123
```

## 小驼峰->连字符

```javascript
const camelToKebab = (str, separator) => {
  return str.replace(/([a-z])([A-Z])/g, `$1${separator}$2`).toLowerCase();

const kebabCase = camelToKebab(camelCase);
```

## 连字符->小驼峰

```javascript
function toCamelCase(str) {
    return str.replace(/[-_](.)/g, function (match, group1) {
        return group1.toUpperCase();
    });
}

console.log(toCamelCase('hello-world')); // helloWorld
console.log(toCamelCase('hello_world')); // helloWorld
```

## 连字符->大驼峰

```javascript
function toUpperCamelCase(str) {
    const words = str.split(/[-_]/);
    const capitalizedWords = words.map(word => word.charAt(0).toUpperCase() + word.slice(1));
    return capitalizedWords.join("");
}

console.log(toUpperCamelCase('hello-world')); // HelloWorld
console.log(toUpperCamelCase('hello_world')); // HelloWorld
```
