# 一维数组

## 一维数组

```javascript
/**
 * 数组交集,一维数组
 * @param {Array} arrOne
 * @param {Array} arrTwo
 * @return {Array}
 */
function arrIntersection(arrOne, arrTwo) {
  return arrOne.filter(function (v) {
    return arrTwo.includes(v);
  });
}

/**
 * 数组差集,一维数组
 * @param {Array} arrOne
 * @param {Array} arrTwo
 * @return {Array}
 * eg: [1, 2, 3] [2, 4, 5] 差集为[1,3,4,5]
 */
function arrDifference(arrOne, arrTwo) {
  return arrOne.concat(arrTwo).filter(function (v) {
    return !arrOne.includes(v) || !arrTwo.includes(v);
  });
}

/**
 * 数组并集,一维数组
 * @param {Array} arrOne
 * @param {Array} arrTwo
 * @return {Array}
 */
function arrAndSet(arrOne, arrTwo) {
  return arrOne.concat(
    arrTwo.filter(function (v) {
      return !arrOne.includes(v);
    })
  );
}

/**
 * 数组补集,一维数组
 * @param {Array} arrOne
 * @param {Array} arrTwo
 * @return {Array}
 */
function arrComplement(arrOne, arrTwo) {
  return arrOne
    .filter(function (v) {
      return !(arrTwo.indexOf(v) > -1);
    })
    .concat(
      arrTwo.filter(function (v) {
        return !(arrOne.indexOf(v) > -1);
      })
    );
}

const arrOne = [1, 2, 3, 4, 5];
const arrTwo = [3, 4, 5, 6, 7];

console.log(arrIntersection(arrOne, arrTwo)); // [ 3, 4, 5 ]
console.log(arrDifference(arrOne, arrTwo)); // [ 1, 2, 6, 7 ]
console.log(arrAndSet(arrOne, arrTwo)); // [ 1, 2, 3, 4, 5, 6, 7 ]
console.log(arrComplement(arrOne, arrTwo)); // [ 1, 2, 6, 7 ]
```
