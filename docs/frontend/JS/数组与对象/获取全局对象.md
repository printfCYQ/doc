# 获取全局对象

## 获取全局对象



## a==1 && a==2 && a==3

```javascript
const a = {
  _default: 0,
  toString: function () {
    return ++this._default;
  },
};

// 对象与数字比较时 --> 对象调用toString方法 --> 返回 '1' --> '1' == 1 成立 --> ... 
if (a == 1 && a == 2 && a == 3) {
  console.log(true);
}
```
```javascript
var _defult = 0;
Object.defineProperty(window, "a", {
  get() {
    return ++_defult;
  },
});

if (a === 1 && a === 2 && a === 3) {
  console.log(true);
}
```

## ([]+[]).length

```javascript
let len1 = ({} + {}).length; // 类型转换 toString ===> 30
console.log({}.toString()); // [object Object]

let len2 = ([] + []).length; // 类型转换 toString ===> 0
console.log([].toString()); // ''
console.log([1, 2, 3].toString()); // '1,2,3'

let len3 = function () {}.length; // 形参个数 ===> 0

function test(a, b, c) {
  console.log(arguments.length); // 实参个数 ===> 4
}
test(1, 2, 3, 4);
console.log(test.length); // 形参个数 ===> 3
```
