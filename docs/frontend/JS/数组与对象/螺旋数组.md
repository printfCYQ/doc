## 螺旋数组

```javascript
function vortex(n, m) {
  const nums = new Array(n).fill(0).map(() => new Array(m).fill(0));
  let i = 0,
    j = 0,
    stepi = 1,
    stepj = 0;
  let count = 1;
  function hasBlock() {
    return !nums[j] || nums[j][i] !== 0;
  }
  while (1) {
    nums[j][i] = count++;
    i += stepi;
    j += stepj;
    // 超出边界 || 已经写值【转弯】
    if (hasBlock()) {
      // 回退一格
      i -= stepi;
      j -= stepj;

      if (stepj === 0) {
        stepj = stepi;
        stepi = 0;
      } else {
        stepi = -stepj;
        stepj = 0;
      }
      i += stepi;
      j += stepj;
    }

    // 结束
    if (hasBlock()) {
      break;
    }
  }
  return nums;
}
```

```javascript
const generateSpiralMatrix = (n, m) => {
  const matrix = Array.from(
    { length: n }, 
    () => Array.from({ length: m }, () => 0)
  );

  let value = 1;
  let topRow = 0;
  let bottomRow = n - 1;
  let leftColumn = 0;
  let rightColumn = m - 1;

  while (topRow <= bottomRow && leftColumn <= rightColumn) {
    // 上行
    for (let i = leftColumn; i <= rightColumn; i++) {
      matrix[topRow][i] = value++;
    }
    topRow++;

    // 右列
    for (let i = topRow; i <= bottomRow; i++) {
      matrix[i][rightColumn] = value++;
    }
    rightColumn--;

    // 下行
    for (let i = rightColumn; i >= leftColumn; i--) {
      matrix[bottomRow][i] = value++;
    }
    bottomRow--;

    // 左列
    for (let i = bottomRow; i >= topRow; i--) {
      matrix[i][leftColumn] = value++;
    }
    leftColumn++;
  }

  return matrix;
};

const n = 6; // 替换为你想要的行数
const m = 5; // 替换为你想要的列数
const spiralMatrix = generateSpiralMatrix(n, m);
```

```javascript
Array.prototype.snail = function (rowsCount, colsCount) {
  if (rowsCount * colsCount !== this.length) return []
  
  const arr = Array.from(
    { length: rowsCount }, 
    () => Array.from({ length: colsCount }, () => 0)
  );
  
  let count = 0
  while (count < this.length) {
    const index = Math.floor(count / rowsCount)
    if (index % 2) {
      for (let i = rowsCount - 1; i >= 0; i--) {
        arr[i][index] = this[count]
        count++
      }
    } else {
      for (let i = 0; i < rowsCount; i++) {
        arr[i][index] = this[count]
        count++
      }
    }
  }
  return arr
}
```
