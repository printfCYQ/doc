# 找出下面数组的质数，去除重复项，按升序排列。

## 找出下面数组的质数，去除重复项，按升序排列。

  

```javascript
    // 找出下面数组的质数，去除重复项，按升序排列。
    arr = [33, 23, 12, 9, 2, 3, 4, 5, 6, 7, 2, 3, 4, 6, 8, 12, 3, 7];
    //ES6  Set 数组去重
    function unique(arr) {
      return Array.from(new Set(arr));
    }
    let arr2 = unique(arr);

    //sort 升序排序
    arr2.sort((a, b) => a - b);

    //filter过滤非质数
    let arr3 = arr2.filter(function (val) {
      for (let i = 2; i < val; i++) {
        if (val % i === 0) {
          return false; //除了1和本身还有其他因素的，则返回false
        }
      }
      return val > 1; //素数是正整数，而且1不是素数。
    });
    console.log(arr3);
    // arr3 = [2, 3, 5, 7, 23]
```


## 判断数字是否为质数

  

```plain
function isPrime(num) {
  if (num == 0) return false;
  if (num == 1) return false;
  if (num == 2 || num == 3) return true;
  if (num % 6 != 1 && num % 6 != 5) return false;
  var temp = Math.sqrt(num);
  for (var i = 5; i <= temp; i += 6) {
    if (num % i == 0 || num % (i + 2) == 0) return false;
  }
  return true;
}
```
