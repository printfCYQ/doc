# 爬贴吧贴子的所有楼层

## 爬贴吧贴子的所有楼层

```rust
use reqwest;
use scraper::{Html, Selector};
use std::fs::File;
use std::io::{self, Write};
use tokio;
use url::Url;

#[tokio::main]
async fn main() -> Result<(), Box<dyn std::error::Error>> {
    // 获取用户输入的 URL
    let mut input_url = String::new();
    println!("请输入初始 URL:");
    io::stdin().read_line(&mut input_url)?;
    let url = input_url.trim().to_string();

    // 解析基础 URL
    let base_url = Url::parse(&url)?;

    let mut data = String::new();

    let mut current_url = url;
    loop {
        // 发送 GET 请求到目标网页
        let response = reqwest::get(&current_url).await?.text().await?;
        println!("Fetched URL: {}", current_url);

        // 解析 HTML 内容
        let document = Html::parse_document(&response);

        // 选择包含内容的元素
        let post_content_main_selector = Selector::parse(".d_post_content_main").unwrap();
        let post_content_selector = Selector::parse(".d_post_content").unwrap();
        let post_tail_wrap_selector = Selector::parse(".post-tail-wrap .tail-info").unwrap();
        let next_page_selector = Selector::parse("a").unwrap();

        // 迭代找到的元素并提取内容
        for el in document.select(&post_content_main_selector) {
            if let Some(content_el) = el.select(&post_content_selector).next() {
                let content = content_el.inner_html().trim().replace("<br>", "\n");
                data.push_str(&content);
                data.push('\n');
            }

            let tail_info_iter: Vec<_> = el.select(&post_tail_wrap_selector).collect();
            match tail_info_iter.len() {
                3 => {
                    if let Some(tail_info1) = tail_info_iter.get(1) {
                        data.push_str(&tail_info1.inner_html());
                    }
                    data.push(' ');
                    if let Some(tail_info2) = tail_info_iter.get(2) {
                        data.push_str(&tail_info2.inner_html());
                    }
                }
                4 => {
                    if let Some(tail_info1) = tail_info_iter.get(2) {
                        data.push_str(&tail_info1.inner_html());
                    }
                    data.push(' ');
                    if let Some(tail_info2) = tail_info_iter.get(3) {
                        data.push_str(&tail_info2.inner_html());
                    }
                }
                _ => {
                    println!(
                        "Unexpected number of tail-info elements: {}",
                        tail_info_iter.len()
                    );
                }
            }
            data.push('\n');
        }

        // 查找“下一页”的链接
        let mut next_page_url = None;
        for element in document.select(&next_page_selector) {
            if let Some(text) = element.text().next() {
                if text.contains("下一页") {
                    if let Some(href) = element.value().attr("href") {
                        next_page_url = Some(href.to_string());
                        break;
                        }
                }
            }
        }

        // 如果找不到“下一页”，则退出循环
        if let Some(next_url) = next_page_url {
            let next_url = base_url.join(&next_url)?;
            current_url = next_url.to_string();
        } else {
            break;
        }
    }

    // 将结果写入文本文件
    let mut file = File::create("result.txt")?;
    file.write_all(data.as_bytes())?;

    println!("结果已保存到 result.txt 文件中");

    Ok(())
}
```
