# 部分可选

## 部分可选

```typescript
interface Article {
  title: string;
  content: string;
  author: string;
  date: Date;
}

type Optional<T, K extends keyof T> = Omit<T, K> & Partial<Pick<T, K>>

type CreateArticleOption = Optional<Article, 'author' | 'date'>

function createArticle(option: CreateArticleOption) {
  console.log(option);
}

const option = {
  title: 'zs',
  content: 'hhh'
}
createArticle(option)
```

## 从字段到函数的推导

```typescript
type Watcher<T> = {
  on<K extends string & keyof T>(
    eventName: `${K}Changed`,
    callback: (oldValue: T[K], newValue: T[K]) => void
  ): void
}

declare function watch<T>(obj:T) : Watcher<T>

const personWatcher = watch({
  name: 'zs',
  age: 18,
  sex: '男'
})

personWatcher.on('ageChanged', (oldValue, newValue) => {
  console.log(oldValue, newValue)
})
```

## infer

### 实现ReturnType

```typescript
/**
 * 表示从函数类型中提取返回类型的类型。
 * 
 * @template T - 函数类型或其他类型
 */
type Return<T> = T extends (...args: any[]) => infer R ? R : T

type sum = (a: number, b: number) => number
type concat = (a: any[], b: any[]) => any[]

let sumResult: Return<sum>; // `number`
let concatResult: Return<concat>; // `any[]`
```

### PromiseType

```typescript
/**
 * 提取 Promise 类型的泛型参数类型。
 * 如果 T 是 Promise 类型，则返回 Promise 泛型参数 K；
 * 否则返回 T 本身。
 *
 * @template T - 要提取泛型参数类型的类型
 */
type PromiseType<T> = T extends Promise<infer K> ? K : T

type pt1 = PromiseType<string> // pt1 的类型是 string
type pt2 = PromiseType<Promise<number>> // pt2 的类型是 number
type pt3 = PromiseType<Promise<Promise<string>>> // pt3 的类型是 Promise<string>

/**
 * 递归地提取嵌套 Promise 类型的泛型参数类型。
 * 如果 T 是 Promise 类型，则继续递归提取 Promise 泛型参数类型；
 * 否则返回 T 本身。
 *
 * @template T - 要提取嵌套 Promise 泛型参数类型的类型
 */
type DeepPromiseType<T> = T extends Promise<infer K> ? DeepPromiseType<K> : T

type pt4 = DeepPromiseType<Promise<Promise<string>>> // pt4 的类型是 string
```

### FirstArg

```typescript
/**
 * 提取函数类型的第一个参数类型。
 * 如果 T 是函数类型，则返回第一个参数的类型 F；
 * 否则返回 T 本身。
 *
 * @template T - 要提取第一个参数类型的函数类型
 */
type FirstArg<T> = T extends (first: infer F, ...args: any[]) => any ? F : T;

type fn1 = FirstArg<() => void>; // fn1 的类型是 unknown
type fn2 = FirstArg<(a: number, b: string) => void>; // fn2 的类型是 number
```

### ArrayType

```typescript
/**
 * 提取数组类型的元素类型。
 * 如果 T 是数组类型，则返回元素类型 I；
 * 否则返回 T 本身。
 *
 * @template T - 要提取元素类型的数组类型
 */
type ArrayType<T> = T extends (infer I)[] ? I : T

type ItemType1 = ArrayType<[string, number]>; // ItemType1 的类型是 string | number
type ItemType2 = ArrayType<string[]>; // ItemType2 的类型是 string
```

## 递归推导类型（curry）柯里化函数

```typescript
/**
 * 表示柯里化函数类型。
 * 
 * @template A - 参数的元组
 * @template R - 返回类型
 */
type Curried<A, R> =
	A extends [] ? () => R : // 如果参数的元组为空，则返回一个没有参数且返回类型为 `R` 的函数
	A extends [infer ARG] ? (param: ARG) => R : // 如果参数的元组只有一个参数，则返回一个接受该参数并返回类型为 `R` 的函数
	A extends [infer ARG, ...infer REST] ? (param: ARG) => Curried<REST, R> : // 如果参数的元组有多个参数，则返回一个接受第一个参数并返回剩余参数柯里化后的函数
	never; // 如果参数的元组不匹配上述情况，则返回 `never`

/**
 * 将普通函数转换为柯里化函数。
 * 
 * @template A - 参数的元组
 * @template R - 返回类型
 * @param {(...args: A) => R} fn - 要转换的普通函数
 * @returns {Curried<A, R>} - 转换后的柯里化函数
 */
declare function curry<A extends any[], R>(fn: (...args: A) => R): Curried<A, R>;

function sum(a: string, b: number, c: object) {
	return a + b + c
}

// 将 `sum` 函数转换为柯里化函数
const currySum = curry(sum)

// 调用柯里化函数，依次传入参数
currySum('aaa')(1)({}) // 返回 'aaa1[object Object]'
```

## 联合类型（｜）交叉类型（&）

```typescript
type Student = {
    age: number;
    score: number;
};
type Teacher = {
    age: number;
    salary: number
};

type Person1 = Student | Teacher;

const person1: Person1 = {
    age: 10,
    score: 10,
    salary: 10
}
person.age
person.score // 类型“Person”上不存在属性“score”。
person.salary // 类型“Person”上不存在属性“salary”。

type Person2 = Student & Teacher;

// 不能将类型“{ age: number; }”分配给类型“Person2”。类型 "{ age: number; }" 中缺少属性 "score"，但类型 "Student" 中需要该属性。
const person2: Person2 = {
    age: 10,
}

// 不能将类型“{ age: number; score: number; }”分配给类型“Person2”。类型 "{ age: number; score: number; }" 中缺少属性 "salary"，但类型 "Teacher" 中需要该属性。
const person3: Person2 = {
    age: 10,
    score: 10,
}
```
> 在代码中，我们定义了 Student 和 Teacher 两种类型，然后使用联合类型 Person1 将它们合并。Person1 类型可以是 Student 或 Teacher。变量 person1 被赋值为一个拥有 age、score 和 salary 属性的对象。由于 Person1 是联合类型，并不知道具体是 Student 还是 Teacher，所以可以访问 age 属性，但无法确定是否可以访问 score 或 salary 属性。
> 
> 接下来，我们定义了交叉类型 Person2，将 Student 和 Teacher 类型合并成一个新的类型。Person2 类型具有 Student 和 Teacher 类型的所有属性。然后，我们试图将一个只有 age 属性的对象赋值给 person2，但由于缺少 score 属性，导致类型不匹配。同样地，我们试图将一个拥有 age 和 score 属性的对象赋值给 person3，但由于缺少 salary 属性，导致类型不匹配
```typescript
type t1 = 'a' & string // "a"

type t2 = 'a' & 'b' // never

type t3 = 'a' | 'b' // "a" | "b"

type t4 = 'a' | 'b' | 1 // "a" | "b" | 1

type t5 = 'a' | 'b' | 1 & string // "a" | "b"
// t5 === t6
type t6 = ('a' & string) | ('b' & string) | (1 & string) // "a" | "b"

type Person = {
    name: string;
    [key: symbol]: string;
};

const person: Person = {
    name: 'aaa',
    [Symbol("id")]: "12345"
}

type PersonKeys = string & keyof Person; // "name"
```
