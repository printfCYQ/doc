# 案例-全局消息提示

## 案例-全局消息提示

> -   编写组件
> -   ​`install`挂载
> -   ​`app.use()`全局注册

```vue
<template>
    <transition name="slide-fade">
        <div v-show="showRef" class="message" :style="{
            backgroundColor: options.backgroundColor
        }">
            <div class="message-content" :style="{
                color: options.textColor
            }">{{ options.message }}</div>
        </div>
    </transition>
</template>
    
<script setup lang='ts'>
const showRef = ref<boolean>(false)
interface Options {
    message?: string,
    duration?: number,
    backgroundColor?: string,
    textColor?: string
    borderColor?: string
}
const options = reactive<Options>({
    message: '提示',
    duration: 3000,
    backgroundColor: '',
    textColor: '',
    borderColor: '',
})
const backgroundColor = {
    primary: '#409EFF',
    success: ' #f0f9eb',
    warning: '#fdf6ec',
    danger: '#fef0f0',
    info: '#edf2fc',
}
const textColor = {
    primary: '#fff',
    success: '#67C23A',
    warning: '#E6A23C',
    danger: '#F56C6C',
    info: '#909399',
}
const borderColor = {
    primary: '#fff',
    success: '#e1f3d8',
    warning: '#fdf6ec',
    danger: '#fde2e2',
    info: '#ebeef5',
}
const primary = (opts?: Options) => {
    options.backgroundColor = opts?.backgroundColor ?? backgroundColor.primary
    options.textColor = opts?.textColor ?? textColor.primary
    options.borderColor = opts?.borderColor ?? borderColor.primary
    showMessage(opts)
}

const success = (opts?: Options) => {
    options.backgroundColor = opts?.backgroundColor ?? backgroundColor.success
    options.textColor = opts?.textColor ?? textColor.success
    options.borderColor = opts?.borderColor ?? borderColor.success
    showMessage(opts)
}

const warning = (opts?: Options) => {
    options.backgroundColor = opts?.backgroundColor ?? backgroundColor.warning
    options.textColor = opts?.textColor ?? textColor.warning
    options.borderColor = opts?.borderColor ?? borderColor.warning
    showMessage(opts)
}

const danger = (opts?: Options) => {
    options.backgroundColor = opts?.backgroundColor ?? backgroundColor.danger
    options.textColor = opts?.textColor ?? textColor.danger
    options.borderColor = opts?.borderColor ?? borderColor.danger
    showMessage(opts)
}

const info = (opts?: Options) => {
    options.backgroundColor = opts?.backgroundColor ?? backgroundColor.info
    options.textColor = opts?.textColor ?? textColor.info
    options.borderColor = opts?.borderColor ?? borderColor.info
    showMessage(opts)
}

const showMessage = (opts?: Options) => {
    options.message = opts?.message ?? options.message
    options.duration = opts?.duration ?? options.duration
    showRef.value = true
    setTimeout(() => {
        showRef.value = false
    }, options.duration)
}
//对外暴露 当前组件的属性和方法
defineExpose({
    primary,
    success,
    danger,
    warning,
    info
})
</script>
```

```typescript
import { createVNode, render, VNode, App } from 'vue';
import Message from './index.vue'
interface Options {
  message?: string,
  duration?: number,
  backgroundColor?: string,
  textColor?: string
  borderColor?: string
}
export default {
  install(app: App) {
    //createVNode vue提供的底层方法 可以给我们组件创建一个虚拟DOM 也就是Vnode
    const vnode: VNode = createVNode(Message)
    //render 把我们的Vnode 生成真实DOM 并且挂载到指定节点
    render(vnode, document.body)
    // Vue 提供的全局配置 可以自定义
    app.config.globalProperties.$Message = {
      primary: (options: Options) => vnode.component?.exposed?.primary(options),
      success: (options: Options) => vnode.component?.exposed?.success(options),
      danger: (options: Options) => vnode.component?.exposed?.danger(options),
      warning: (options: Options) => vnode.component?.exposed?.warning(options),
      info: (options: Options) => vnode.component?.exposed?.info(options)
    }
  }
}
```
```typescript
import { createApp } from 'vue';
import App from './App.vue';
import Message from './components/Message';
const app = createApp(App)

app.use(Message)

interface Options {
    message?: string,
    duration?: number,
    backgroundColor?: string,
    textColor?: string
    borderColor?: string
}
type MessageType = {
    primary: (options?: Options) => void
    success: (options?: Options) => void
    danger: (options?: Options) => void
    warning: (options?: Options) => void
    info: (options?: Options) => void
}
//编写ts loading 声明文件放置报错 和 智能提示
declare module 'vue' {
    export interface ComponentCustomProperties {
        $Message: MessageType
    }
}

app.mount('#app')
```

```typescript
<template>
  <div>
    <button @click="clickBtn('primary')">primary</button>
    <button @click="clickBtn('success')">success</button>
    <button @click="clickBtn('danger')">danger</button>
    <button @click="clickBtn('warning')">warning</button>
    <button @click="clickBtn('info')">info</button>
  </div>
</template>

<script setup lang="ts">
import { getCurrentInstance } from 'vue';

const instance = getCurrentInstance()
const clickBtn = (type: string) => {
  instance?.proxy?.$Message[type]()
}
</script>
```
