## 模板

```tsx
const renderDom = () => {
  return (
    <>
      <div>TSX</div>
    </>
  );
};

export default renderDom;
```
```vue
<template>
  <div>
    <C></C>
  </div>
</template>

<script setup lang="ts">
  import C from './C'
</script>
```

## v-model

> 需要`.value`

```tsx
import { ref } from "vue";

let v = ref<string>("");

const renderDom = () => {
  return (
    <>
      <input v-model={v.value} type="text" />
      <div>{v.value}</div>
    </>
  );
};

export default renderDom;
```

## v-show

```tsx
import { ref } from "vue";

let flag = ref(false);

const renderDom = () => {
  return (
    <>
      <div v-show={flag.value}>true</div>
      <div v-show={!flag.value}>false</div>
    </>
  );
};

export default renderDom;
```

## v-if

> 三元表达式

```tsx
import { ref } from "vue";

let flag = ref(false);

const renderDom = () => {
  return <>{flag.value ? <div>true</div> : <div>false</div>}</>;
};

export default renderDom;
```

## v-for

> map

```tsx
let arr = [1, 2, 3, 4, 5];

const renderDom = () => {
  return (
    <>
      {arr.map((v) => {
        return <div>{v}</div>;
      })}
    </>
  );
};

export default renderDom;
```

## v-bind

```tsx
let arr = [1, 2, 3, 4, 5];
import D from "./D";
const renderDom = () => {
  return (
    <>
      <D arr={arr}>1</D>
    </>
  );
};

export default renderDom;
```

## props

```tsx
const renderDom = (props: any) => {
  return (
    <>
      <div>
        {props["arr"].map((v: number) => (
          <div>{v}</div>
        ))}
      </div>
    </>
  );
};

export default renderDom;
```

## v-on

> `React`风格

```tsx
const renderDom = () => {
  return (
    <>
      <button onClick={clickTap}>点击</button>
    </>
  );
};

const clickTap = () => {
  console.log("click");
};

export default renderDom;
```

## emit

```tsx
import D from "./D";
const renderDom = () => {
  return (
    <>
      <D
        title={123}
        onClick={(params: number) => {
          getMsg(params);
        }}
      ></D>
    </>
  );
};

const getMsg = (params: number) => {
  console.log(params);
};
export default renderDom;
```
```tsx
const renderDom = (props: any, content: any) => {
  const clickTap = () => {
    content.emit("click", 1);
  };
  return (
    <>
      <div>{props.title}</div>
      <button
        onClick={() => {
          clickTap();
        }}
      >
        点击
      </button>
    </>
  );
};

export default renderDom;
```
