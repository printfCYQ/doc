# 生产环境不转变

> vite 在打包时【生产环境】 小图片会自动打包成base64

## 生产环境不转变

```javascript
import vue from '@vitejs/plugin-vue'
import { defineConfig } from 'vite'

// https://vitejs.dev/config/
export default defineConfig({
  plugins: [vue()],
  build: {
    assetsInlineLimit: 0, // 默认4096 小于4096 会被base64编码
  }
})

```



## 插件-开发环境也转base64

```javascript
import vue from '@vitejs/plugin-vue';
import fs from 'fs';
import { defineConfig } from 'vite';

const imgTransformPlugin = (limit: number = 4096) => {
  return {
    name: 'img-transform',
    async transform(code: string, id: string) {
      // 只处理开发环境
      if (process.env.NODE_ENV !== 'development') {
        return
      }
      // 只处理图片
      if (!id.endsWith('.png')) {
        return
      }

      const stat = await fs.promises.stat(id)
      console.log(stat);
      if (stat.size > limit) {
        return
      }
      const base64 = fs.readFileSync(id, { encoding: 'base64' })
      const dataUrl = `data:image/png;base64,${base64}`
      return {
        code: `export default "${dataUrl}"`
      }
    }
  }
}

// https://vitejs.dev/config/
export default defineConfig({
  plugins: [vue(), imgTransformPlugin()],
  build: {
    assetsInlineLimit: 4096,
  }
})

```
