# 借助父组件（复杂，不推荐）

## 借助父组件（复杂，不推荐）

```html
<template>
  <div>
    <A @on-click="getMsg"></A>
    <B :info="info"></B>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import A from './A.vue';
import B from './B.vue';
let info = ref<string>('')
const getMsg = (msg: string) => {
  info.value = msg
}
</script>
```
```html
<template>
    <div>
        <button @click="sendMsg">sendMsg</button>
    </div>
</template>

<script setup lang="ts">
const emit = defineEmits(['on-click'])

const sendMsg = () => {
    emit('on-click', '我是消息')
}
</script>
```
```html
<template>
    <div>
        {{ info }}
    </div>
</template>

<script setup lang="ts">

defineProps<{
    info: string
}>()
</script>
```

## EventBus

> 发布订阅模式
> 
> 简单实现

```typescript

type BusClass<T> = {
  emit: (name: T) => void
  on: (name: T, callback: Function) => void
}
type BusParams = string | number | symbol 
type List = {
  [key: BusParams]: Array<Function>
}
class Bus<T extends BusParams> implements BusClass<T> {
  list: List
  constructor() {
    this.list = {}
  }
  emit(name: T, ...args: Array<any>) {
    let eventName: Array<Function> = this.list[name]
    eventName.forEach(ev => {
      ev.apply(this, args)
    })
  }
  on(name: T, callback: Function) {
    let fn: Array<Function> = this.list[name] || [];
    fn.push(callback)
    this.list[name] = fn
  }
}

export default new Bus<string>()
```
```typescript
<template>
    <div>
        <button @click="sendMsg">sendMsg</button>
    </div>
</template>

<script setup lang="ts">
import Bus from '../utils/bus'
const sendMsg = () => {
    Bus.emit('on-click', '消息')
}
</script>
```
```typescript
<template>
    <div>
        {{ info }}
    </div>
</template>

<script setup lang="ts">
import { ref } from 'vue';
import Bus from '../utils/bus'
const info = ref('')

```

## Mitt

> 完善的EventBus。

`pnpm i mitt -S`

> ​[https://github.com/developit/mitt](https://github.com/developit/mitt)

### 一般使用

```typescript
import mitt from 'mitt';
import { createApp } from 'vue';
import App from './App.vue';

const Mit = mitt()
declare module 'vue' {
    export interface ComponentCustomProperties {
        $Bus: typeof Mit
    }
}
const app = createApp(App)
app.config.globalProperties.$Bus = Mit
app.mount('#app')

```
```typescript
<template>
    <div>
        <button @click="sendMsg">sendMsg</button>
    </div>
</template>

<script setup lang="ts">
import { getCurrentInstance } from 'vue'
const instance = getCurrentInstance()
const sendMsg = () => {
    instance?.proxy?.$Bus.emit('on-test', '消息')
}
</script>
```
```typescript
<template>
    <div>
        {{ info }}
    </div>
</template>

<script setup lang="ts">
import { getCurrentInstance, ref } from 'vue'
const instance = getCurrentInstance()
const info = ref('')
instance?.proxy?.$Bus.on('on-test', (str: any) => {
    info.value = str
})
</script>
```

### 其他方法

```typescript
// 监听所有方法
// type:事件名称
// str:参数
instance?.proxy?.$Bus.on('*', (type, str) => {
    console.log(type, str);
})
```
```typescript
// 删除指定事件
instance?.proxy?.$Bus.off('on-test', (e) => {
    console.log(e);
})

// 删除所有事件监听
instance?.proxy?.$Bus.all.clear()
```
