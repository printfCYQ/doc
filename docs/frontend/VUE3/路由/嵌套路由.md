# 嵌套路由

> 一些应用程序的 UI 由多层嵌套的组件组成。在这种情况下，URL 的片段通常对应于特定的嵌套组件结构，例如：

```typescript
const routes: Array<RouteRecordRaw> = [{
    path: '/home',
    name: 'Home',
    component: () => import('@/components/A.vue'),
    children: [
        {
            path: 'about',
            name: 'About',
            component: () => import('@/components/B.vue')
        },
        {
            path: 'info',
            name: 'Info',
            component: () => import('@/components/C.vue')
        }
    ]
}]
```
```typescript
<template>
    <div>
        AAA
        <router-view></router-view>
    </div>
</template>
```
```typescript
<template>
  <div>
    <p>
      <router-link to="/home/about">Go to About</router-link>
      <router-link to="/home/info">Go to Info</router-link>
    </p>
    <router-view></router-view>
  </div>
</template>
```
